const express = require('express'); const session = require('express-session'); const path = require('path'); const app = express(); const PORT = 3000; // 미들웨어 설정 app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, 'public'))); // 세션 설정 (DB 없이 메모리 세션 사용) app.use(session({ secret: 'my-secret-key', resave: false, saveUninitialized: false, cookie: { maxAge: 60 * 60 * 1000 } // 1시간 유지 })); // 하드코딩된 임시 계정 정보 (DB 대용) const MOCK_USER = { username: 'admin', password: 'password123', name: '관리자' }; // 로그인 API app.post('/api/login', (req, res) => { const { username, password } = req.body; if (username === MOCK_USER.username && password === MOCK_USER.password) { req.session.user = { username: MOCK_USER.username, name: MOCK_USER.name }; return res.json({ success: true, message: '로그인 성공', user: req.session.user }); } else { return res.status(401).json({ success: false, message: '아이디 또는 비밀번호가 올바르지 않습니다.' }); } }); // 로그인 상태 확인 API app.get('/api/user', (req, res) => { if (req.session.user) { res.json({ loggedIn: true, user: req.session.user }); } else { res.json({ loggedIn: false }); } }); // 로그아웃 API app.post('/api/logout', (req, res) => { req.session.destroy((err) => { if (err) { return res.status(500).json({ success: false, message: '로그아웃 실패' }); } res.clearCookie('connect.sid'); res.json({ success: true, message: '로그아웃 성공' }); }); }); app.listen(PORT, () => { console.log(`서버가 실행되었습니다: http://localhost:${PORT}`); });
